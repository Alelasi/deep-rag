### 1. BGE-M3 多功能嵌入模型 ⭐ 2026 SOTA

**核心特性**：Multi-Linguality + Multi-Functionality + Multi-Granularity

#### 三大"M"能力

```
Multi-Linguality（多语言）
├─ 支持 100+ 语言
├─ 跨语言语义对齐
└─ 中英文混合查询

Multi-Functionality（多功能）
├─ Dense Retrieval（稠密检索）
├─ Sparse Retrieval（稀疏检索）
└─ Multi-Vector Retrieval（多向量检索）

Multi-Granularity（多粒度）
├─ 短句：<128 tokens
├─ 段落：128-512 tokens
└─ 长文档：512-8192 tokens
```

#### 实现

```python
from FlagEmbedding import BGEM3FlagModel

class BGEM3Retriever:
    """BGE-M3 多功能检索器"""
    
    def __init__(self):
        self.model = BGEM3FlagModel('BAAI/bge-m3', use_fp16=True)
    
    def encode(self, texts: list[str], return_dense=True, return_sparse=True, return_colbert=False):
        """多模式编码
        
        Returns:
            dense: (N, 1024) 稠密向量
            sparse: dict 稀疏向量（类似 SPLADE）
            colbert: (N, seq_len, 1024) 多向量（可选）
        """
        embeddings = self.model.encode(
            texts,
            return_dense=return_dense,
            return_sparse=return_sparse,
            return_colbert_vecs=return_colbert,
            max_length=8192  # 支持长文档
        )
        
        return embeddings
    
    def hybrid_search(self, query: str, corpus: list[str], top_k: int = 10):
        """混合搜索（Dense + Sparse）"""
        # 编码
        query_emb = self.encode([query])
        corpus_emb = self.encode(corpus)
        
        # Dense 检索
        dense_scores = query_emb['dense_vecs'] @ corpus_emb['dense_vecs'].T
        
        # Sparse 检索（词汇匹配）
        sparse_scores = self._compute_sparse_scores(
            query_emb['lexical_weights'],
            corpus_emb['lexical_weights']
        )
        
        # 融合（可调权重）
        final_scores = 0.5 * dense_scores + 0.5 * sparse_scores
        
        # 排序
        top_k_idx = np.argsort(final_scores[0])[::-1][:top_k]
        return top_k_idx
    
    def _compute_sparse_scores(self, query_weights, corpus_weights):
        """计算稀疏向量分数"""
        scores = []
        for doc_weights in corpus_weights:
            score = sum(
                query_weights.get(token, 0) * doc_weights.get(token, 0)
                for token in set(query_weights) & set(doc_weights)
            )
            scores.append(score)
        return np.array(scores)
```

### 2. SPLADE 学习式稀疏检索 🆕 超越 BM25

**核心思想**：神经网络学习稀疏表示，比传统 BM25 更智能

#### 原理

```
传统 BM25:
"深度学习" → [深度:1.2, 学习:1.5]（仅匹配原词）

SPLADE:
"深度学习" → [深度:1.2, 学习:1.5, 神经网络:0.8, AI:0.6, 机器学习:0.7]
                ↑ 自动扩展相关词
```

#### 实现

```python
from transformers import AutoModelForMaskedLM, AutoTokenizer
import torch

class SPLADERetriever:
    """SPLADE 学习式稀疏检索"""
    
    def __init__(self, model_name="naver/splade-cocondenser-ensembledistil"):
        self.model = AutoModelForMaskedLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model.eval()
    
    def encode(self, texts: list[str]):
        """编码为稀疏向量"""
        inputs = self.tokenizer(
            texts,
            return_tensors="pt",
            padding=True,
            truncation=True,
            max_length=512
        )
        
        with torch.no_grad():
            outputs = self.model(**inputs)
            logits = outputs.logits
            
            # ReLU + log(1 + x) 激活
            sparse_vecs = torch.log(1 + torch.relu(logits))
            
            # Max pooling（每个 token 取最大值）
            sparse_vecs = torch.max(
                sparse_vecs * inputs["attention_mask"].unsqueeze(-1),
                dim=1
            )[0]
        
        return sparse_vecs
    
    def search(self, query: str, corpus_vecs: torch.Tensor, top_k: int = 10):
        """稀疏向量搜索"""
        query_vec = self.encode([query])
        
        # 点积（稀疏向量相似度）
        scores = torch.matmul(query_vec, corpus_vecs.T)[0]
        
        # 排序
        top_k_idx = torch.argsort(scores, descending=True)[:top_k]
        return top_k_idx.tolist()
```

### 3. Late Chunking（晚期分块）🆕 上下文保留

**核心思想**：先编码整个文档，再分块，保留跨块上下文

#### 传统 vs Late Chunking

```
传统分块（Early Chunking）:
文档 → 分块 → 逐块编码
  ↓
"深度学习" 和 "神经网络" 在不同块
  ↓
丢失上下文关系

Late Chunking:
文档 → 整体编码 → 分块
  ↓
"深度学习" 和 "神经网络" 共享上下文
  ↓
保留语义关系
```

#### 实现

```python
from sentence_transformers import SentenceTransformer
import numpy as np

class LateChunkingRetriever:
    """Late Chunking 检索器"""
    
    def __init__(self, model_name="jinaai/jina-embeddings-v2-base-en"):
        self.model = SentenceTransformer(model_name)
        self.chunk_size = 512  # tokens
    
    def encode_with_late_chunking(self, document: str):
        """Late Chunking 编码"""
        # 1. 整体编码（保留上下文）
        full_embedding = self.model.encode(
            document,
            convert_to_tensor=True,
            output_value='token_embeddings'  # 返回 token 级别
        )  # (seq_len, dim)
        
        # 2. 分块（基于 token 位置）
        chunks = self._split_tokens(document, self.chunk_size)
        
        # 3. 为每个块提取对应的 embedding
        chunk_embeddings = []
        start_idx = 0
        for chunk in chunks:
            chunk_len = len(self.model.tokenizer.encode(chunk))
            end_idx = start_idx + chunk_len
            
            # 平均池化该块的 token embeddings
            chunk_emb = full_embedding[start_idx:end_idx].mean(dim=0)
            chunk_embeddings.append(chunk_emb)
            
            start_idx = end_idx
        
        return torch.stack(chunk_embeddings)
    
    def _split_tokens(self, text: str, chunk_size: int):
        """分块（保留句子完整性）"""
        sentences = text.split('。')
        chunks = []
        current_chunk = []
        current_len = 0
        
        for sent in sentences:
            sent_len = len(self.model.tokenizer.encode(sent))
            if current_len + sent_len > chunk_size and current_chunk:
                chunks.append('。'.join(current_chunk) + '。')
                current_chunk = [sent]
                current_len = sent_len
            else:
                current_chunk.append(sent)
                current_len += sent_len
        
        if current_chunk:
            chunks.append('。'.join(current_chunk) + '。')
        
        return chunks
```

### 4. 生产级缓存策略 ⚡ 延迟 -80%

**核心思想**：两层缓存 + 智能预热

#### 架构

```
查询 → L1: Redis (Embedding Cache)
          ↓ 命中率 60%
       L2: Vector Index (HNSW/IVF)
          ↓
       L3: 原始文档
```

#### 实现

```python
import redis
import hashlib
import pickle

class CachedRetriever:
    """带缓存的检索器"""
    
    def __init__(self, retriever, redis_client):
        self.retriever = retriever
        self.redis = redis_client
        self.embedding_ttl = 3600  # 1 小时
        self.result_ttl = 300       # 5 分钟
    
    def retrieve(self, query: str, k: int = 10):
        """带缓存的检索"""
        # 1. 查询结果缓存（L1）
        result_key = f"result:{self._hash(query)}:{k}"
        cached_result = self.redis.get(result_key)
        if cached_result:
            return pickle.loads(cached_result)
        
        # 2. Embedding 缓存（L2）
        emb_key = f"emb:{self._hash(query)}"
        cached_emb = self.redis.get(emb_key)
        
        if cached_emb:
            query_emb = pickle.loads(cached_emb)
        else:
            # 编码查询
            query_emb = self.retriever.encode(query)
            self.redis.setex(emb_key, self.embedding_ttl, pickle.dumps(query_emb))
        
        # 3. 向量检索
        results = self.retriever.search(query_emb, k=k)
        
        # 4. 缓存结果
        self.redis.setex(result_key, self.result_ttl, pickle.dumps(results))
        
        return results
    
    def _hash(self, text: str) -> str:
        """生成缓存键"""
        return hashlib.md5(text.encode()).hexdigest()
    
    def warmup(self, common_queries: list[str]):
        """预热缓存"""
        for query in common_queries:
            self.retrieve(query)
```

### 5. Multi-Vector Retrieval（多向量检索）🆕 细粒度匹配

**核心思想**：一个文档 → 多个向量，更精细的匹配

#### 原理

```
传统（Single-Vector）:
文档 → 1 个向量 (1024 维)
  ↓
粗粒度匹配

Multi-Vector:
文档 → N 个向量 (每个段落/句子 1 个)
  ↓
细粒度匹配（类似 ColBERT）
```

#### 实现

```python
class MultiVectorRetriever:
    """多向量检索器"""
    
    def __init__(self, model):
        self.model = model
    
    def encode_document(self, document: str):
        """文档 → 多个向量"""
        # 分段（按句子）
        sentences = document.split('。')
        
        # 每个句子编码为一个向量
        vectors = self.model.encode(sentences)
        
        return vectors  # (N, dim)
    
    def search(self, query: str, doc_vectors_list: list, top_k: int = 10):
        """多向量搜索"""
        query_vec = self.model.encode([query])[0]
        
        scores = []
        for doc_vectors in doc_vectors_list:
            # MaxSim：query 与文档中最相似的向量
            sims = np.dot(doc_vectors, query_vec)
            max_sim = sims.max()
            scores.append(max_sim)
        
        # 排序
        top_k_idx = np.argsort(scores)[::-1][:top_k]
        return top_k_idx
```

### 7. 异步优化循环 🔄 零停机重建

**核心思想**：后台异步重建索引，不影响查询

#### 实现

```python
import threading
import time

class AsyncOptimizer:
    """异步索引优化器"""
    
    def __init__(self, index):
        self.index = index
        self.running = False
    
    def start_optimization_loop(self, interval_hours: int = 24):
        """启动优化循环"""
        self.running = True
        
        def optimize():
            while self.running:
                # 1. 创建新索引（后台）
                new_index = self._rebuild_index()
                
                # 2. 原子切换（零停机）
                self._atomic_swap(new_index)
                
                # 3. 清理旧索引
                self._cleanup_old_index()
                
                # 4. 等待下一轮
                time.sleep(interval_hours * 3600)
        
        thread = threading.Thread(target=optimize, daemon=True)
        thread.start()
    
    def _rebuild_index(self):
        """重建索引（优化 HNSW 图）"""
        # 从数据库加载所有向量
        vectors = self.load_all_vectors()
        
        # 创建新索引
        new_index = faiss.IndexHNSWFlat(self.dim, M=32)
        new_index.hnsw.efConstruction = 200
        new_index.add(vectors)
        
        return new_index
    
    def _atomic_swap(self, new_index):
        """原子切换索引"""
        # 使用锁保证原子性
        with self.lock:
            old_index = self.index
            self.index = new_index
        
        return old_index
```
